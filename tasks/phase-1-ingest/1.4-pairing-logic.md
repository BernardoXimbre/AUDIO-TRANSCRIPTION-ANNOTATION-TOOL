```typescript
import * as path from 'path';

export interface PairItem {
  audio: string;
  transcript: string;
}

export interface PairingResult {
  matched: PairItem[];
  unmatchedAudio: string[];
  unmatchedTranscript: string[];
}

/**
 * Extract base filename without extension
 * @param filename e.g., "sample_clean.wav" → "sample_clean"
 */
function getBaseName(filename: string): string {
  return path.basename(filename, path.extname(filename)).toLowerCase();
}

/**
 * Auto-pair audio and transcript files by matching base filenames
 * 
 * @param audioFilenames Array of audio filenames (e.g., ["a.wav", "b.mp3"])
 * @param transcriptFilenames Array of transcript filenames (e.g., ["a.json", "b.json"])
 * @returns PairingResult with matched pairs and unmatched items
 * 
 * @example
 * autoPair(["a.wav", "b.mp3", "c.wav"], ["a.json", "d.json"])
 * // Returns:
 * // {
 * //   matched: [{ audio: "a.wav", transcript: "a.json" }],
 * //   unmatchedAudio: ["b.mp3", "c.wav"],
 * //   unmatchedTranscript: ["d.json"]
 * // }
 */
export function autoPair(
  audioFilenames: string[],
  transcriptFilenames: string[]
): PairingResult {
  const matched: PairItem[] = [];
  const unmatchedAudio = new Set(audioFilenames);
  const unmatchedTranscript = new Set(transcriptFilenames);
  
  // Build map of base names to transcript filenames
  const transcriptMap = new Map<string, string>();
  transcriptFilenames.forEach(filename => {
    const baseName = getBaseName(filename);
    transcriptMap.set(baseName, filename);
  });
  
  // Try to match each audio file
  audioFilenames.forEach(audioFile => {
    const baseName = getBaseName(audioFile);
    const matchedTranscript = transcriptMap.get(baseName);
    
    if (matchedTranscript) {
      matched.push({
        audio: audioFile,
        transcript: matchedTranscript
      });
      unmatchedAudio.delete(audioFile);
      unmatchedTranscript.delete(matchedTranscript);
    }
  });
  
  return {
    matched,
    unmatchedAudio: Array.from(unmatchedAudio),
    unmatchedTranscript: Array.from(unmatchedTranscript)
  };
}

/**
 * Validate pairing - ensure no duplicates
 * @param pairing Array of matched pairs
 * @throws Error if duplicates found
 */
export function validatePairing(pairing: PairItem[]): void {
  const audioUsed = new Set<string>();
  const transcriptUsed = new Set<string>();
  
  pairing.forEach(({ audio, transcript }) => {
    if (audioUsed.has(audio)) {
      throw new Error(\`Audio file "\${audio}" paired multiple times\`);
    }
    if (transcriptUsed.has(transcript)) {
      throw new Error(\`Transcript file "\${transcript}" paired multiple times\`);
    }
    audioUsed.add(audio);
    transcriptUsed.add(transcript);
  });
}

/**
 * Apply manual overrides to pairing result
 * @param initial Pairing result from autoPair
 * @param overrides Array of manual pairs { audio, transcript }
 * @returns Updated PairingResult
 */
export function applyManualOverrides(
  initial: PairingResult,
  overrides: PairItem[]
): PairingResult {
  const result = { ...initial };
  
  overrides.forEach(override => {
    // Remove from unmatched sets
    result.unmatchedAudio = result.unmatchedAudio.filter(a => a !== override.audio);
    result.unmatchedTranscript = result.unmatchedTranscript.filter(t => t !== override.transcript);
    
    // Add to matched (or replace if already matched)
    const existingIndex = result.matched.findIndex(m => m.audio === override.audio);
    if (existingIndex >= 0) {
      result.matched[existingIndex] = override;
    } else {
      result.matched.push(override);
    }
  });
  
  // Re-validate
  validatePairing(result.matched);
  
  return result;
}
```

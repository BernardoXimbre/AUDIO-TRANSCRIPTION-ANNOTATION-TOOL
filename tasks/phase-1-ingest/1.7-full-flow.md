```typescript
import { prisma } from '../db';
import { extractAudioMetadata } from '../utils/audioMetadata';
import { autoPair, applyManualOverrides, PairItem } from './pairingService';
import * as fs from 'fs';
import * as path from 'path';

const UPLOAD_DIR = process.env.AUDIO_UPLOAD_DIR || './uploads';
const AUTO_REJECT_DURATION_SECONDS = 15;

export async function processIngest(
  audioFileObjects: any[], // multer file objects
  transcriptJSON: any,
  manualPairings: PairItem[] = []
) {
  const results = {
    ingested: 0,
    rejected: 0,
    unmatched: { audio: [] as string[], transcripts: [] as string[] },
    errors: [] as string[]
  };
  
  try {
    // Extract audio filenames
    const audioFilenames = audioFileObjects.map(f => f.originalname);
    
    // Parse transcripts
    let transcripts: any[];
    try {
      transcripts = typeof transcriptJSON === 'string' 
        ? JSON.parse(transcriptJSON) 
        : transcriptJSON;
      if (!Array.isArray(transcripts)) {
        throw new Error('Transcripts must be an array');
      }
    } catch (error) {
      results.errors.push(\`Invalid transcript JSON: \${error instanceof Error ? error.message : String(error)}\`);
      throw new Error('Invalid transcript JSON');
    }
    
    const transcriptFilenames = transcripts.map(t => t.path);
    
    // Auto-pair
    let pairing = autoPair(audioFilenames, transcriptFilenames);
    
    // Apply manual overrides
    if (manualPairings.length > 0) {
      pairing = applyManualOverrides(pairing, manualPairings);
    }
    
    // Ingest matched pairs
    for (const pair of pairing.matched) {
      try {
        const audioFileObj = audioFileObjects.find(f => f.originalname === pair.audio);
        const transcriptData = transcripts.find(t => t.path === pair.transcript);
        
        if (!audioFileObj || !transcriptData) {
          results.errors.push(\`Missing file for pair: \${pair.audio} ↔ \${pair.transcript}\`);
          continue;
        }
        
        // Extract audio metadata
        const metadata = await extractAudioMetadata(audioFileObj.path);
        const { duration, sampleRate, channels, bitDepth } = metadata;
        
        // Check 15-second rule
        const isAutoRejected = duration < AUTO_REJECT_DURATION_SECONDS;
        
        // Sanitize filename (prevent path traversal)
        const sanitizedFilename = sanitizeFilename(pair.audio);
        
        // Move audio file to permanent location
        const permanentPath = path.join(UPLOAD_DIR, 'permanent', sanitizedFilename);
        await ensureDir(path.dirname(permanentPath));
        fs.copyFileSync(audioFileObj.path, permanentPath);
        fs.unlinkSync(audioFileObj.path); // Remove temp file
        
        // Create AudioFile record
        const audioFile = await prisma.audioFile.create({
          data: {
            filename: sanitizedFilename,
            path: permanentPath,
            duration,
            sampleRate,
            channels,
            bitDepth,
            fileSize: audioFileObj.size,
            bextMetadata: null, // TODO: Extract from WAV if present
            listInfoMetadata: null
          }
        });
        
        // Create Transcript record
        const transcript = await prisma.transcript.create({
          data: {
            audioFileId: audioFile.id,
            originalText: transcriptData.label,
            correctedText: transcriptData.label, // Initially same as original
            isAutoRejected,
            status: isAutoRejected ? 'rejected' : 'pending'
          }
        });
        
        // Create Recording record with calculated values
        const speechRateWPM = calculateSpeechRate(transcriptData.label, duration);
        const distanceEstimate = estimateDistance(audioFileObj); // Placeholder
        
        await prisma.recording.create({
          data: {
            audioFileId: audioFile.id,
            speechRate: speechRateWPM,
            distanceEstimate,
            distanceMethod: 'rms_level_analysis', // Placeholder
            rmsLevel: null, // TODO: Calculate from audio
            peakLevel: null,
            noiseFloor: null
          }
        });
        
        results.ingested++;
        if (isAutoRejected) results.rejected++;
        
      } catch (error) {
        const errorMsg = error instanceof Error ? error.message : String(error);
        results.errors.push(\`Failed to ingest \${pair.audio}: \${errorMsg}\`);
      }
    }
    
    // Record unmatched items
    results.unmatched = {
      audio: pairing.unmatchedAudio,
      transcripts: pairing.unmatchedTranscript
    };
    
    return results;
  } catch (error) {
    const errorMsg = error instanceof Error ? error.message : String(error);
    results.errors.push(errorMsg);
    throw error;
  }
}

/**
 * Sanitize filename to prevent path traversal
 */
function sanitizeFilename(filename: string): string {
  return filename
    .replace(/\.\./g, '')
    .replace(/\\\\/g, '/')
    .replace(/[\x00-\x1F\x7F]/g, '')
    .replace(/^\.+/, ''); // Remove leading dots
}

/**
 * Ensure directory exists
 */
async function ensureDir(dirPath: string): Promise<void> {
  if (!fs.existsSync(dirPath)) {
    fs.mkdirSync(dirPath, { recursive: true });
  }
}

/**
 * Calculate speech rate (words per minute)
 */
function calculateSpeechRate(text: string, durationSeconds: number): number {
  if (durationSeconds === 0) return 0;
  const wordCount = text.split(/\s+/).length;
  const durationMinutes = durationSeconds / 60;
  return Math.round(wordCount / durationMinutes);
}
```

```typescript
import { Request, Response } from 'express';
import { processIngest } from '../services/ingestService';

export async function ingestAudio(req: Request, res: Response) {
  try {
    const audioFiles = (req.files as Express.Multer.File[]) || [];
    const transcriptJSON = req.body.transcripts;
    const manualPairings = req.body.manualPairings || [];
    
    if (audioFiles.length === 0) {
      return res.status(400).json({ error: 'No audio files provided' });
    }
    
    if (!transcriptJSON) {
      return res.status(400).json({ error: 'No transcript JSON provided' });
    }
    
    const result = await processIngest(audioFiles, transcriptJSON, manualPairings);
    
    res.status(result.errors.length > 0 ? 207 : 200).json({
      success: result.errors.length === 0,
      ingested: result.ingested,
      rejected: result.rejected,
      unmatched: result.unmatched,
      errors: result.errors
    });
  } catch (error) {
    const message = error instanceof Error ? error.message : String(error);
    res.status(400).json({
      success: false,
      error: message,
      errors: [message]
    });
  }
}
```

## ⚡ Exactly

Create `apps/backend/src/controllers/ingestController.ts`:

```typescript
import { Request, Response } from 'express';
import { prisma } from '../db';
import { ingestService } from '../services/ingestService';

export async function ingestAudio(req: Request, res: Response) {
  try {
    // Files will be in req.files (after multer middleware)
    // Transcript JSON in req.body.transcripts
    
    const result = await ingestService.processIngest(
      req.files || [],
      req.body.transcripts || []
    );
    
    res.status(200).json({
      success: true,
      audioFiles: result.audioFiles,
      transcripts: result.transcripts,
      autoRejectedCount: result.autoRejectedCount
    });
  } catch (error) {
    res.status(500).json({ 
      error: String(error),
      status: 500
    });
  }
}

export async function getQueue(req: Request, res: Response) {
  try {
    const { status, minDuration, maxDuration } = req.query;
    
    let query: any = {
      isAutoRejected: false // Exclude auto-rejected
    };
    
    if (status) query.status = status;
    if (minDuration || maxDuration) {
      query.audioFile = {
        duration: {}
      };
      if (minDuration) query.audioFile.duration.gte = parseFloat(minDuration);
      if (maxDuration) query.audioFile.duration.lte = parseFloat(maxDuration);
    }
    
    const items = await prisma.transcript.findMany({
      where: query,
      include: {
        audioFile: {
          select: { filename: true, duration: true, filepath: true }
        }
      },
      orderBy: { createdAt: 'desc' }
    });
    
    res.status(200).json(items);
  } catch (error) {
    res.status(500).json({ error: String(error) });
  }
}
```

**Add to Express app** (`server.ts`):

```typescript
import multer from 'multer';
import { ingestAudio, getQueue } from './controllers/ingestController';

const upload = multer({ 
  dest: process.env.AUDIO_UPLOAD_DIR || './uploads',
  limits: { fileSize: 100 * 1024 * 1024 } // 100MB
});

app.post('/api/ingest', upload.any(), ingestAudio);
app.get('/api/queue', getQueue);
```

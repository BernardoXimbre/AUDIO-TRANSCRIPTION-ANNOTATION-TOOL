Add to Editor.vue:

```vue
<!-- Corrected Text with Overlaid Spans -->
<div class="relative">
  <!-- Hidden text for positioning -->
  <div class="whitespace-pre-wrap text-sm font-mono invisible">{{ correctedText }}</div>
  
  <!-- Overlay spans -->
  <div class="absolute inset-0 whitespace-pre-wrap text-sm font-mono pointer-events-none">
    <span 
      v-for="annotation in sortedAnnotations"
      :key="annotation.id"
      :class="getSpanClass(annotation.type)"
      :style="getSpanStyle(annotation)"
      @mouseenter="hoveredAnnotation = annotation"
      @mouseleave="hoveredAnnotation = null"
      @click.stop="selectAnnotation(annotation)"
      class="cursor-pointer transition-opacity hover:opacity-80"
    >
      {{ correctedText.slice(annotation.startOffset, annotation.endOffset) }}
    </span>
  </div>
</div>

<!-- Annotation Details Tooltip -->
<div 
  v-if="hoveredAnnotation"
  class="mt-2 p-2 bg-white border border-gray-300 rounded text-xs"
>
  <p class="font-semibold">{{ hoveredAnnotation.type }}</p>
  <p class="text-gray-600">{{ JSON.stringify(hoveredAnnotation.attributes) }}</p>
</div>

<!-- Annotation List -->
<div v-if="annotations.length > 0" class="mt-4">
  <h4 class="font-semibold mb-2">Annotations ({{ annotations.length }})</h4>
  <div class="space-y-1">
    <div 
      v-for="ann in annotations"
      :key="ann.id"
      class="flex items-center justify-between p-2 bg-gray-50 rounded text-xs"
      :class="getSpanClass(ann.type)"
    >
      <span>{{ ann.type }} ({{ ann.startOffset }}-{{ ann.endOffset }})</span>
      <div class="flex gap-1">
        <button @click="editAnnotation(ann)" class="text-blue-500 hover:text-blue-700">Edit</button>
        <button @click="deleteAnnotation(ann.id)" class="text-red-500 hover:text-red-700">Delete</button>
      </div>
    </div>
  </div>
</div>
```

Add to script:

```typescript
const annotations = ref<any[]>([]);
const hoveredAnnotation = ref<any>(null);

async function loadAnnotations() {
  try {
    const response = await fetch(\`/api/annotation/\${props.transcriptId}\`);
    annotations.value = await response.json();
  } catch (error) {
    console.error('Error loading annotations:', error);
  }
}

function getSpanClass(type: string): string {
  const classes: { [key: string]: string } = {
    'CRUD': 'bg-gray-300',
    'NUMBER': 'bg-blue-300',
    'FORMATTING_COMMAND': 'bg-green-300',
    'SPELLED_OUT': 'bg-purple-300',
    'NAMED_ENTITY': 'bg-orange-300',
    'MEDICAL_TERM': 'bg-red-300',
    'MEASUREMENT': 'bg-yellow-300'
  };
  return classes[type] || 'bg-gray-200';
}

function getSpanStyle(annotation: any) {
  return {
    zIndex: Math.floor(annotation.startOffset / 10) % 10
  };
}

async function deleteAnnotation(id: string) {
  try {
    await fetch(\`/api/annotation/\${id}\`, { method: 'DELETE' });
    annotations.value = annotations.value.filter(a => a.id !== id);
  } catch (error) {
    console.error('Error deleting:', error);
  }
}

const sortedAnnotations = computed(() => {
  return [...annotations.value].sort((a, b) => a.startOffset - b.startOffset);
});
```

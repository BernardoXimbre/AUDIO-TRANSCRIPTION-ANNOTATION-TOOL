## ⚡ Exactly

Implement full CRUD workflow in Annotator.vue:

```typescript
const props = defineProps<{
  transcriptId: string;
}>();

// States
const selectedText = ref('');
const selectedType = ref('');
const attributes = ref<any>({});
const annotations = ref<any[]>([]);
const editingAnnotation = ref<any>(null);

// CREATE
async function createAnnotation() {
  try {
    const response = await fetch('/api/annotation', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        transcriptId: props.transcriptId,
        type: selectedType.value,
        startOffset: startOffset.value,
        endOffset: endOffset.value,
        attributes: attributes.value
      })
    });
    
    if (!response.ok) throw new Error('Failed to create');
    
    const annotation = await response.json();
    annotations.value.push(annotation);
    resetForm();
  } catch (error) {
    console.error('Error creating annotation:', error);
    alert('Error creating annotation. Check console.');
  }
}

// EDIT
async function updateAnnotation() {
  if (!editingAnnotation.value) return;
  
  try {
    const response = await fetch(\`/api/annotation/\${editingAnnotation.value.id}\`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ attributes: attributes.value })
    });
    
    if (!response.ok) throw new Error('Failed to update');
    
    const index = annotations.value.findIndex(a => a.id === editingAnnotation.value.id);
    if (index >= 0) {
      annotations.value[index] = await response.json();
    }
    resetForm();
  } catch (error) {
    console.error('Error updating annotation:', error);
  }
}

// DELETE
async function deleteAnnotation(id: string) {
  try {
    const response = await fetch(\`/api/annotation/\${id}\`, {
      method: 'DELETE'
    });
    
    if (!response.ok) throw new Error('Failed to delete');
    
    annotations.value = annotations.value.filter(a => a.id !== id);
  } catch (error) {
    console.error('Error deleting annotation:', error);
  }
}

// SELECT for editing
function selectAnnotation(annotation: any) {
  editingAnnotation.value = annotation;
  selectedType.value = annotation.type;
  attributes.value = { ...annotation.attributes };
}

// RESET form
function resetForm() {
  selectedText.value = '';
  selectedType.value = '';
  attributes.value = {};
  editingAnnotation.value = null;
}

// LOAD annotations on mount
onMounted(async () => {
  try {
    const response = await fetch(\`/api/annotation/\${props.transcriptId}\`);
    annotations.value = await response.json();
  } catch (error) {
    console.error('Error loading annotations:', error);
  }
});
```

Add to template:

```vue
<!-- Create / Edit Section -->
<div v-if="selectedType" class="mb-4 p-4 bg-blue-50 border border-blue-300 rounded">
  <h3 class="font-semibold mb-2">{{ editingAnnotation ? 'Edit' : 'Create' }} Annotation</h3>
  
  <!-- Type-specific forms here (from Task 3.5) -->
  
  <div class="flex gap-2 mt-4">
    <button 
      @click="editingAnnotation ? updateAnnotation() : createAnnotation()"
      class="px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600"
    >
      {{ editingAnnotation ? 'Update' : 'Create' }}
    </button>
    <button 
      @click="resetForm"
      class="px-4 py-2 bg-gray-300 text-gray-700 rounded hover:bg-gray-400"
    >
      Cancel
    </button>
  </div>
</div>
```

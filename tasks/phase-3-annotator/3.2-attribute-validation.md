export function validateAnnotationAttributes(type: string, attributes: any): { valid: boolean; errors: string[] } {
  const errors: string[] = [];
  
  switch (type) {
    case 'CRUD':
      if (Object.keys(attributes).length > 0) {
        errors.push('CRUD requires empty attributes object');
      }
      break;
      
    case 'NUMBER':
      if (attributes.rendering === undefined || !['digits', 'words'].includes(attributes.rendering)) {
        errors.push('NUMBER requires rendering: "digits" or "words"');
      }
      if (attributes.value === undefined || typeof attributes.value !== 'number') {
        errors.push('NUMBER requires value: number');
      }
      break;
      
    case 'FORMATTING_COMMAND':
      const validCommands = ['newline', 'paragraph', 'period', 'comma', 'dash', 'colon', 'bracket_open', 'bracket_close'];
      if (!validCommands.includes(attributes.command)) {
        errors.push(\`FORMATTING_COMMAND requires command: one of [\${validCommands.join(', ')}]\`);
      }
      if (typeof attributes.isCommand !== 'boolean') {
        errors.push('FORMATTING_COMMAND requires isCommand: boolean');
      }
      break;
      
    case 'SPELLED_OUT':
      if (typeof attributes.resolvedWord !== 'string') {
        errors.push('SPELLED_OUT requires resolvedWord: string');
      }
      break;
      
    case 'NAMED_ENTITY':
      const validCategories = ['person', 'org', 'place', 'date'];
      if (!validCategories.includes(attributes.category)) {
        errors.push(\`NAMED_ENTITY requires category: one of [\${validCategories.join(', ')}]\`);
      }
      break;
      
    case 'MEDICAL_TERM':
      const medicalCategories = ['anatomy', 'procedure', 'diagnosis', 'drug', 'device'];
      if (!medicalCategories.includes(attributes.category)) {
        errors.push(\`MEDICAL_TERM requires category: one of [\${medicalCategories.join(', ')}]\`);
      }
      // note is optional
      break;
      
    case 'MEASUREMENT':
      if (typeof attributes.value !== 'number') {
        errors.push('MEASUREMENT requires value: number');
      }
      const validUnits = ['g', 'mg', 'ug', 'kg', 'ml', 'l', 'mmHg', 'IE', 'mm', 'cm', 'Ch'];
      if (!validUnits.includes(attributes.unit)) {
        errors.push(\`MEASUREMENT requires unit: one of [\${validUnits.join(', ')}]\`);
      }
      if (typeof attributes.normalizedValue !== 'number') {
        errors.push('MEASUREMENT requires normalizedValue: number');
      }
      if (typeof attributes.normalizedUnit !== 'string') {
        errors.push('MEASUREMENT requires normalizedUnit: string');
      }
      break;
      
    default:
      errors.push(\`Unknown annotation type: \${type}\`);
  }
  
  return {
    valid: errors.length === 0,
    errors
  };
}

export function normalizeUnit(value: number, unit: string): { value: number; unit: string } {
  const conversions: { [key: string]: { baseFactor: number; baseUnit: string } } = {
    'mg': { baseFactor: 0.001, baseUnit: 'g' },
    'ug': { baseFactor: 0.000001, baseUnit: 'g' },
    'kg': { baseFactor: 1000, baseUnit: 'g' },
    'ml': { baseFactor: 0.001, baseUnit: 'l' },
    'mm': { baseFactor: 0.1, baseUnit: 'cm' }
  };
  
  if (conversions[unit]) {
    const conv = conversions[unit];
    return {
      value: value * conv.baseFactor,
      unit: conv.baseUnit
    };
  }
  
  return { value, unit };
}

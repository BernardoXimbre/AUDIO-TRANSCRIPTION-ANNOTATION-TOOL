## ⚡ Exactly

Create `apps/frontend/src/components/Editor.vue`:

```vue
<template>
  <div class="editor grid grid-cols-2 gap-4 p-4">
    <!-- Original (Read-only) -->
    <div>
      <h3 class="text-sm font-bold mb-2 text-gray-600">Original Text</h3>
      <div class="p-3 bg-gray-100 rounded text-sm leading-relaxed text-gray-700 cursor-not-allowed pointer-events-none select-none">
        {{ originalText }}
      </div>
    </div>
    
    <!-- Corrected (Editable) -->
    <div>
      <h3 class="text-sm font-bold mb-2 text-gray-600">Corrected Text</h3>
      <textarea 
        v-model="correctedText"
        class="w-full p-3 border border-blue-300 rounded text-sm leading-relaxed font-mono focus:outline-none focus:border-blue-600"
        rows="10"
      ></textarea>
      <div class="text-xs text-gray-500 mt-1">Position: {{ cursorPosition }} / {{ correctedText.length }}</div>
    </div>
  </div>
  
  <!-- Save Button -->
  <div class="p-4 border-t flex justify-end gap-2">
    <button 
      @click="cancel"
      class="px-4 py-2 bg-gray-300 text-gray-700 rounded hover:bg-gray-400"
    >
      Cancel
    </button>
    <button 
      @click="save"
      :disabled="correctedText === originalText"
      class="px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600 disabled:bg-gray-300"
    >
      Save Changes
    </button>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue';

const props = defineProps<{
  transcriptId: string;
}>();

const emit = defineEmits<{
  'saved': [correctedText: string];
}>();

const originalText = ref('');
const correctedText = ref('');
const cursorPosition = ref(0);

async function loadTranscript() {
  try {
    const response = await fetch(\`/api/transcript/\${props.transcriptId}\`);
    const data = await response.json();
    originalText.value = data.originalText;
    correctedText.value = data.correctedText;
  } catch (error) {
    console.error('Error loading transcript:', error);
  }
}

async function save() {
  try {
    const response = await fetch(\`/api/transcript/\${props.transcriptId}\`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ correctedText: correctedText.value })
    });
    
    if (!response.ok) throw new Error('Save failed');
    
    emit('saved', correctedText.value);
  } catch (error) {
    console.error('Error saving:', error);
  }
}

function cancel() {
  correctedText.value = originalText.value;
}

function updateCursorPosition(e: Event) {
  const textarea = e.target as HTMLTextAreaElement;
  cursorPosition.value = textarea.selectionStart;
}

onMounted(() => {
  loadTranscript();
});
</script>

<style scoped>
.editor {
  background: white;
}
</style>
```

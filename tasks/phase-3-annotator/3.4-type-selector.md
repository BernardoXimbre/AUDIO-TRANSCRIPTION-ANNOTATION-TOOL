## ⚡ Exactly

Create `apps/frontend/src/components/Annotator.vue`:

```vue
<template>
  <div class="annotator p-4 max-w-2xl">
    <div v-if="selectedText" class="mb-4 p-3 bg-yellow-50 border border-yellow-300 rounded">
      <p class="text-sm font-semibold">Selected Text:</p>
      <p class="font-mono text-sm">{{ selectedText }}</p>
      <p class="text-xs text-gray-600">Offset: {{ startOffset }} - {{ endOffset }}</p>
    </div>
    
    <div v-if="selectedText" class="mb-4">
      <label class="block text-sm font-semibold mb-2">Annotation Type:</label>
      <select v-model="selectedType" class="border rounded px-2 py-1 w-full">
        <option value="">-- Select Type --</option>
        <option value="CRUD">CRUD (Raw text)</option>
        <option value="NUMBER">NUMBER (Spoken numbers)</option>
        <option value="FORMATTING_COMMAND">FORMATTING_COMMAND (Punctuation/structure)</option>
        <option value="SPELLED_OUT">SPELLED_OUT (Letter-by-letter words)</option>
        <option value="NAMED_ENTITY">NAMED_ENTITY (Person, org, place)</option>
        <option value="MEDICAL_TERM">MEDICAL_TERM (Medical concepts)</option>
        <option value="MEASUREMENT">MEASUREMENT (Units: mg, ml, etc)</option>
      </select>
      
      <div v-if="selectedType" class="mt-2 p-2 bg-blue-50 text-xs text-blue-700 rounded">
        {{ getTypeDescription(selectedType) }}
      </div>
    </div>
    
    <button 
      @click="proceedToForm"
      :disabled="!selectedText || !selectedType"
      class="px-4 py-2 bg-green-500 text-white rounded hover:bg-green-600 disabled:bg-gray-300"
    >
      Continue to Attributes →
    </button>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue';

const props = defineProps<{
  correctedText: string;
}>();

const emit = defineEmits<{
  'type-selected': [type: string, text: string, offsets: { start: number; end: number }];
}>();

const selectedText = ref('');
const startOffset = ref(0);
const endOffset = ref(0);
const selectedType = ref('');

function handleTextSelection() {
  const selection = window.getSelection();
  if (!selection || selection.toString().length === 0) {
    selectedText.value = '';
    return;
  }
  
  selectedText.value = selection.toString();
  // Simplified offset calculation (production would need better handling)
  startOffset.value = 0;
  endOffset.value = selectedText.value.length;
}

function getTypeDescription(type: string): string {
  const descriptions: { [key: string]: string } = {
    'CRUD': 'Keep as-is: raw transcription text',
    'NUMBER': 'Spoken numbers (e.g., "twenty-five" → 25)',
    'FORMATTING_COMMAND': 'Punctuation/structure commands',
    'SPELLED_OUT': 'Letter-by-letter spelled words',
    'NAMED_ENTITY': 'Person names, organizations, places, dates',
    'MEDICAL_TERM': 'Medical concepts, procedures, drugs',
    'MEASUREMENT': 'Numeric values with units (mg, ml, etc)'
  };
  return descriptions[type] || '';
}

function proceedToForm() {
  emit('type-selected', selectedType.value, selectedText.value, {
    start: startOffset.value,
    end: endOffset.value
  });
}

onMounted(() => {
  document.addEventListener('mouseup', handleTextSelection);
  return () => document.removeEventListener('mouseup', handleTextSelection);
});
</script>

<style scoped>
.annotator {
  background: white;
}
</style>
```

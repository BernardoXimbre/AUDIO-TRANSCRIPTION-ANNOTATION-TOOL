## ⚡ Exactly

Create `apps/backend/src/controllers/annotationController.ts`:

```typescript
import { Request, Response } from 'express';
import { prisma } from '../db';

export async function createAnnotation(req: Request, res: Response) {
  try {
    const { transcriptId, type, startOffset, endOffset, attributes } = req.body;
    
    if (!transcriptId || !type || startOffset === undefined || endOffset === undefined) {
      return res.status(400).json({ error: 'Missing required fields' });
    }
    
    const annotation = await prisma.annotation.create({
      data: {
        transcriptId,
        type,
        startOffset,
        endOffset,
        attributes: attributes || {}
      }
    });
    
    res.status(201).json(annotation);
  } catch (error) {
    res.status(400).json({ error: String(error) });
  }
}

export async function getAnnotations(req: Request, res: Response) {
  try {
    const { transcriptId } = req.params;
    
    const annotations = await prisma.annotation.findMany({
      where: { transcriptId }
    });
    
    res.json(annotations);
  } catch (error) {
    res.status(500).json({ error: String(error) });
  }
}

export async function updateAnnotation(req: Request, res: Response) {
  try {
    const { id } = req.params;
    const { attributes } = req.body;
    
    const annotation = await prisma.annotation.update({
      where: { id },
      data: { attributes }
    });
    
    res.json(annotation);
  } catch (error) {
    res.status(400).json({ error: String(error) });
  }
}

export async function deleteAnnotation(req: Request, res: Response) {
  try {
    const { id } = req.params;
    
    await prisma.annotation.delete({ where: { id } });
    
    res.status(204).send();
  } catch (error) {
    res.status(400).json({ error: String(error) });
  }
}
```

Add to Express app:

```typescript
app.post('/api/annotation', createAnnotation);
app.get('/api/annotation/:transcriptId', getAnnotations);
app.patch('/api/annotation/:id', updateAnnotation);
app.delete('/api/annotation/:id', deleteAnnotation);
```

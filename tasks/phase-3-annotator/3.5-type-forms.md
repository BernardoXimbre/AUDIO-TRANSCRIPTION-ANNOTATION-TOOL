  </div>
  
  <!-- FORMATTING_COMMAND Form -->
  <div v-else-if="selectedType === 'FORMATTING_COMMAND'" class="space-y-2">

  <div v-else-if="selectedType === 'NAMED_ENTITY'" class="space-y-2">

    <div>

const attributes = ref<any>({});

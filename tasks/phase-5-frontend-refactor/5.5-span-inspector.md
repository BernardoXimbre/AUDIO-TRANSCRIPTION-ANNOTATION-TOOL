            ? 'bg-white text-primary border-b-2 border-primary'
            : 'text-slate-600 hover:text-slate-900'
        ]"
      >
        Recording (4.6)
      </button>
    </div>

    <!-- CONTENT -->
    <div class="flex-1 flex flex-col overflow-auto p-3">
      <!-- TAB: SPAN INSPECTOR -->
      <div v-if="activeTab === 'span'" class="flex flex-col gap-3">
        <!-- Selected Text Display -->
        <div v-if="selectedAnnotation" class="p-2 bg-sky-50 rounded border border-sky-200">
          <p class="text-xs text-slate-600 font-medium mb-1">Selected Text:</p>
          <p class="text-xs font-mono text-slate-800 font-bold">
            {{ selectedAnnotation.text }}
          </p>
        </div>

        <!-- Type Selector (6 buttons) -->
        <div class="flex flex-col gap-2">
          <p class="text-xs font-semibold text-slate-700">Annotation Type *</p>
          <div class="grid grid-cols-2 gap-1">
            <button
              v-for="type in annotationTypes"
              :key="type.value"
              @click="selectedType = type.value"
              :class="[
                'py-1.5 px-2 rounded text-xs font-semibold transition-all',
                selectedType === type.value
                  ? 'bg-primary text-white ring-2 ring-primary/50'
                  : 'bg-slate-100 text-slate-700 hover:bg-slate-200'
              ]"
              :title="`${type.label} (Alt+${type.hotkey})`"
            >
              {{ type.label }} <span class="text-[10px] opacity-75">{{ type.hotkey }}</span>
            </button>
          </div>
        </div>

        <!-- Type-Specific Attributes -->
        <div v-if="selectedType" class="border-t border-border pt-3">
          <!-- MEASUREMENT attributes -->
          <div v-if="selectedType === 'MEASUREMENT'" class="flex flex-col gap-2">
            <p class="text-xs font-semibold text-slate-700">Measurement Details</p>

            <div>
              <label class="text-xs text-slate-600">Value *</label>
              <input
                v-model.number="measurementValue"
                type="number"
                class="w-full text-xs px-2 py-1 border border-border rounded"
                placeholder="e.g., 1500"
              />
            </div>

            <div>
              <label class="text-xs text-slate-600">Unit *</label>
              <select v-model="measurementUnit" class="w-full text-xs px-2 py-1 border border-border rounded">

<script setup lang="ts">
import { ref, computed } from 'vue';
import { useAnnotationStore } from '@/store/annotationStore';

const store = useAnnotationStore();
const activeTab = ref<'span' | 'recording'>('span');
const selectedType = ref<string | null>(null);
const measurementValue = ref<number | null>(null);
const measurementUnit = ref<string>('');
const speechRate = ref<number | null>(null);
const distanceEstimate = ref<string>('medium');

const annotationTypes = [
  { value: 'CRUD', label: 'CRUD', hotkey: '1' },
  { value: 'NUMBER', label: 'NUMBER', hotkey: '2' },
  { value: 'FORMATTING_COMMAND', label: 'FORMAT', hotkey: '3' },
  { value: 'SPELLED_OUT', label: 'SPELLED', hotkey: '4' },
  { value: 'NAMED_ENTITY', label: 'ENTITY', hotkey: '5' },
  { value: 'MEDICAL_TERM', label: 'MEDICAL', hotkey: '6' },
  { value: 'MEASUREMENT', label: 'MEASURE', hotkey: '7' }
];

const selectedAnnotation = computed(() => store.selectedAnnotation);

const audioMetadata = computed(() => {
  const t = store.transcripts.find(tr => tr.id === store.currentTranscriptId);
  return t?.audioFile;
});

const normalizedValue = computed(() => {
  if (!measurementValue.value || !measurementUnit.value) return '--';

  // Simple SI normalization
  const conversions: Record<string, { to: string; factor: number }> = {
    'mg': { to: 'g', factor: 0.001 },
    'ml': { to: 'l', factor: 0.001 },
    'mm': { to: 'cm', factor: 0.1 }
  };

  const conv = conversions[measurementUnit.value];
  if (!conv) return `${measurementValue.value} ${measurementUnit.value}`;

  return (measurementValue.value * conv.factor).toFixed(2);
});

const normalizedUnit = computed(() => {
  const units: Record<string, string> = {
    'mg': 'g',
    'ml': 'l',
    'mm': 'cm'
  };
  return units[measurementUnit.value] || measurementUnit.value;
});

const confirmSpan = async () => {
  if (!selectedType.value) return;

  // TODO: Save annotation to backend
  console.log('Confirming span:', {
    type: selectedType.value,
    value: measurementValue.value,
    unit: measurementUnit.value
  });

  // Clear form
  selectedType.value = null;
  measurementValue.value = null;
  measurementUnit.value = '';
};

const discardSpan = () => {
  selectedType.value = null;
  measurementValue.value = null;
  measurementUnit.value = '';
};

const deleteSpan = async () => {
  if (!selectedAnnotation.value?.id) return;

  // TODO: Delete annotation from backend
  console.log('Deleting span:', selectedAnnotation.value.id);
};

// GLOBAL HOTKEY LISTENER for Alt+1..7
if (typeof window !== 'undefined') {
  window.addEventListener('keydown', (e: KeyboardEvent) => {
    if (e.altKey) {
      const typeMap: Record<string, string> = {
        '1': 'CRUD',
        '2': 'NUMBER',
        '3': 'FORMATTING_COMMAND',
        '4': 'SPELLED_OUT',
        '5': 'NAMED_ENTITY',
        '6': 'MEDICAL_TERM',
        '7': 'MEASUREMENT'
      };

      if (typeMap[e.key]) {
        e.preventDefault();
        selectedType.value = typeMap[e.key];
      }
    }
  });
}
</script>

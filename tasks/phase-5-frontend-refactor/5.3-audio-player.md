## 🏗️ Implementation

### Create/Update: `apps/frontend/src/components/AudioPlayer.vue`

```vue
<template>
  <div class="p-3 border-b border-border bg-white flex flex-col gap-2">
    <!-- Playback Time + Seek Slider -->
    <div class="flex flex-col gap-1">
      <div class="flex items-center justify-between text-xs font-mono">
        <span class="font-bold">{{ formatTime(currentTime) }}</span>
        <span class="text-slate-500">/</span>
        <span>{{ formatTime(duration) }}</span>
      </div>

      <!-- Simple Range Slider (no canvas) -->
      <input
        v-model="currentTime"
        type="range"
        :min="0"
        :max="duration"
        class="w-full h-1 bg-slate-200 rounded-lg appearance-none cursor-pointer accent-primary"
        @change="onSeek"
      />
    </div>

    <!-- CONTROLS: -2s / Play / +2s / Loop -->
    <div class="flex items-center justify-center gap-2">
      <!-- -2s [J] -->
      <button
        @click="rewind"
        class="px-2 py-1 bg-white border border-border rounded hover:bg-slate-50 flex items-center gap-1 text-xs"
        title="Jump -2s (J)"
      >
        ⏪ -2s <kbd class="text-xs bg-slate-100 px-1 rounded">J</kbd>
      </button>

      <!-- Play/Pause [Space] -->
      <button
        @click="togglePlayPause"
        :class="[
          'px-4 py-1 rounded font-semibold text-white transition-colors',
          isPlaying ? 'bg-red-500 hover:bg-red-600' : 'bg-primary hover:bg-primary-hover'
        ]"
        title="Play/Pause (Space)"
      >
        {{ isPlaying ? '⏸ Pause' : '▶ Play' }}
        <kbd class="text-xs bg-white/20 px-1 rounded ml-1">Space</kbd>
      </button>

      <!-- +2s [L] -->
      <button
        @click="forward"
        class="px-2 py-1 bg-white border border-border rounded hover:bg-slate-50 flex items-center gap-1 text-xs"
        title="Jump +2s (L)"
      >
        +2s ⏩ <kbd class="text-xs bg-slate-100 px-1 rounded">L</kbd>
      </button>

      <!-- Loop Span [R] -->
      <button
        @click="toggleLoopSpan"
        :class="[
          'px-2 py-1 rounded flex items-center gap-1 text-xs transition-colors',
          loopActive
            ? 'bg-primary text-white'
            : 'bg-white border border-border hover:bg-slate-50 text-slate-700'
        ]"
        title="Loop active span (R)"
      >
        🔁 Loop Span <kbd class="text-xs bg-slate-100 px-1 rounded">R</kbd>
      </button>
    </div>

    <!-- SPEED TOGGLES -->
    <div class="flex items-center justify-center gap-2">
      <span class="text-xs font-medium text-slate-600">Speed:</span>
      <div class="flex items-center bg-slate-100 rounded border border-border">
        <button
          v-for="speed in [0.75, 1, 1.25, 1.5, 2]"
          :key="speed"
          @click="playbackSpeed = speed"
          :class="[
            'px-2 py-1 text-xs transition-colors',
            playbackSpeed === speed
              ? 'bg-white text-primary font-bold border-r border-border'
              : 'text-slate-600 hover:text-slate-900'
          ]"
        >
          {{ speed }}x
        </button>
      </div>
    </div>

    <!-- Hidden Audio Element -->
    <audio
      ref="audioElement"
      @timeupdate="onTimeUpdate"
      @ended="onAudioEnded"
      @play="isPlaying = true"
      @pause="isPlaying = false"
    ></audio>
  </div>
</template>

<script setup lang="ts">
import { ref, computed, watch } from 'vue';
import { useAnnotationStore } from '@/store/annotationStore';

const store = useAnnotationStore();
const audioElement = ref<HTMLAudioElement | null>(null);

const currentTime = ref(0);
const duration = ref(0);
const isPlaying = ref(false);
const playbackSpeed = ref(1);
const loopActive = ref(false);
const loopStart = ref(0);
const loopEnd = ref(0);

// Load audio from current transcript
watch(
  () => store.currentTranscriptId,
  async (transcriptId) => {
    if (!transcriptId || !audioElement.value) return;

    const transcript = store.transcripts.find(t => t.id === transcriptId);
    if (transcript) {
      audioElement.value.src = `/uploads/audio/permanent/${transcript.audioFile.filename}`;
      audioElement.value.load();
    }
  },
  { immediate: true }
);

// Sync playback speed
watch(playbackSpeed, (speed) => {
  if (audioElement.value) {
    audioElement.value.playbackRate = speed;
  }
});

// Sync current time to store
watch(currentTime, (time) => {
  store.playbackTime = time;
});

const formatTime = (seconds: number) => {
  const mins = Math.floor(seconds / 60);
  const secs = (seconds % 60).toFixed(2);
  return `${mins}:${secs.padStart(5, '0')}`;
};

const togglePlayPause = () => {
  if (!audioElement.value) return;

  if (isPlaying.value) {
    audioElement.value.pause();
  } else {
    audioElement.value.play();
  }
};

const rewind = () => {
  if (audioElement.value) {
    audioElement.value.currentTime = Math.max(0, audioElement.value.currentTime - 2);
  }
};

const forward = () => {
  if (audioElement.value) {
    audioElement.value.currentTime = Math.min(duration.value, audioElement.value.currentTime + 2);
  }
};

const toggleLoopSpan = () => {
  loopActive.value = !loopActive.value;
  // TODO: Loop active annotation span
};

const onSeek = () => {
  if (audioElement.value) {
    audioElement.value.currentTime = currentTime.value;
  }
};

const onTimeUpdate = () => {
  if (audioElement.value) {
    currentTime.value = audioElement.value.currentTime;
    duration.value = audioElement.value.duration;

    // Handle loop
    if (loopActive.value && currentTime.value >= loopEnd.value) {
      audioElement.value.currentTime = loopStart.value;
    }
  }
};

const onAudioEnded = () => {
  if (audioElement.value) {
    audioElement.value.currentTime = 0;
  }
};

// GLOBAL HOTKEY LISTENER
const handleKeyDown = (e: KeyboardEvent) => {
  if (e.code === 'Space' && (e.target as HTMLElement).tagName !== 'INPUT') {
    e.preventDefault();
    togglePlayPause();
  } else if (e.key === 'j' || e.key === 'J') {
    e.preventDefault();
    rewind();
  } else if (e.key === 'l' || e.key === 'L') {
    e.preventDefault();
    forward();
  } else if (e.key === 'r' || e.key === 'R') {
    e.preventDefault();
    toggleLoopSpan();
  }
};

if (typeof window !== 'undefined') {
  window.addEventListener('keydown', handleKeyDown);
}
</script>
```

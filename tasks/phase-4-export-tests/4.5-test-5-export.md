```typescript
import { describe, test, expect, beforeAll, afterAll } from '@jest/globals';
import { PrismaClient } from '@prisma/client';
import fetch from 'node-fetch';

const prisma = new PrismaClient();
const API_URL = process.env.API_URL || 'http://localhost:5000';

describe('Export JSONL Schema', () => {
  
  let createdTranscriptId: string;
  
  beforeAll(async () => {
    // Create test data
    const audioFile = await prisma.audioFile.create({
      data: {
        filename: 'test_export.wav',
        path: '/test/test_export.wav',
        duration: 30.0,
        sampleRate: 16000,
        channels: 1,
        bitDepth: 16,
        fileSize: 960000
      }
    });
    
    const transcript = await prisma.transcript.create({
      data: {
        audioFileId: audioFile.id,
        originalText: 'Test original text',
        correctedText: 'Test corrected text',
        isAutoRejected: false,
        status: 'completed'
      }
    });
    
    await prisma.recording.create({
      data: {
        audioFileId: audioFile.id,
        speechRate: 150,
        distanceEstimate: 'close'
      }
    });
    
    await prisma.annotation.create({
      data: {
        transcriptId: transcript.id,
        type: 'MEDICAL_TERM',
        startOffset: 5,
        endOffset: 15,
        attributes: { category: 'procedure' }
      }
    });
    
    createdTranscriptId = transcript.id;
  });
  
  test('Export JSONL has correct schema', async () => {
    const response = await fetch(\`\${API_URL}/api/export\`, {
      method: 'POST'
    });
    
    expect(response.ok).toBe(true);
    const content = await response.text();
    
    const lines = content.split('\\n').filter(l => l.trim());
    expect(lines.length).toBeGreaterThan(0);
    
    // Parse each line as JSON
    for (const line of lines) {
      const obj = JSON.parse(line); // Must be valid JSON
      
      // Required fields
      expect(obj).toHaveProperty('audioPath');
      expect(obj).toHaveProperty('duration');
      expect(obj).toHaveProperty('recordingConditions');
      expect(obj).toHaveProperty('transcript');
      expect(obj).toHaveProperty('annotations');
      
      // Type checks
      expect(typeof obj.audioPath).toBe('string');
      expect(typeof obj.duration).toBe('number');
      expect(Array.isArray(obj.annotations)).toBe(true);
      
      // Recording conditions
      expect(obj.recordingConditions).toHaveProperty('sampleRate');
      expect(obj.recordingConditions).toHaveProperty('speechRate');
      expect(obj.recordingConditions).toHaveProperty('distanceEstimate');
      
      // Transcript
      expect(obj.transcript).toHaveProperty('original');
      expect(obj.transcript).toHaveProperty('corrected');
      expect(typeof obj.transcript.original).toBe('string');
      expect(typeof obj.transcript.corrected).toBe('string');
      
      // Annotations structure
      for (const ann of obj.annotations) {
        expect(ann).toHaveProperty('type');
        expect(ann).toHaveProperty('startOffset');
        expect(ann).toHaveProperty('endOffset');
        expect(ann).toHaveProperty('attributes');
        
        expect(typeof ann.type).toBe('string');
        expect(typeof ann.startOffset).toBe('number');
        expect(typeof ann.endOffset).toBe('number');
        expect(typeof ann.attributes).toBe('object');
      }
    }
  });
  
  test('Export content type is JSONL', async () => {
    const response = await fetch(\`\${API_URL}/api/export\`, {
      method: 'POST'
    });
    
    const contentType = response.headers.get('content-type');
    expect(contentType).toBe('application/jsonl');
  });
  
  test('Only completed transcripts exported', async () => {
    // Create a pending transcript that should NOT be exported
    const audioFile = await prisma.audioFile.create({
      data: {
        filename: 'not_completed.wav',
        path: '/test/not_completed.wav',
        duration: 20.0,
        sampleRate: 16000,
        channels: 1,
        bitDepth: 16,
        fileSize: 640000
      }
    });
    
    await prisma.transcript.create({
      data: {
        audioFileId: audioFile.id,
        originalText: 'Pending text',
        correctedText: 'Pending text',
        isAutoRejected: false,
        status: 'pending' // NOT completed
      }
    });
    
    const response = await fetch(\`\${API_URL}/api/export\`, {
      method: 'POST'
    });
    
    const content = await response.text();
    expect(content).not.toContain('Pending text');
  });
});
```

```typescript
import { describe, test, expect, beforeAll, afterAll } from '@jest/globals';
import { PrismaClient } from '@prisma/client';
import * as fs from 'fs';
import * as path from 'path';

const prisma = new PrismaClient();

describe('15-Second Auto-Rejection Rule', () => {
  
  test('Audio < 15s is auto-rejected and NOT in queue', async () => {
    // Create a short audio file (mock or use test file)
    const testAudioPath = path.join(__dirname, '../../data/test/short_10sec.wav');
    
    // Create AudioFile record
    const audioFile = await prisma.audioFile.create({
      data: {
        filename: 'short_10sec.wav',
        path: testAudioPath,
        duration: 10.5, // Less than 15 seconds
        sampleRate: 16000,
        channels: 1,
        bitDepth: 16,
        fileSize: 336000 // Typical for 10s mono 16-bit
      }
    });
    
    // Create Transcript (should be auto-rejected)
    const transcript = await prisma.transcript.create({
      data: {
        audioFileId: audioFile.id,
        originalText: 'This is a short clip',
        correctedText: 'This is a short clip',
        isAutoRejected: true, // Should be set by ingest
        status: 'rejected' // Should NOT be pending
      }
    });
    
    // Verify isAutoRejected = true
    expect(transcript.isAutoRejected).toBe(true);
    expect(transcript.status).toBe('rejected');
    
    // Query queue - should NOT include this transcript
    const queueItems = await prisma.transcript.findMany({
      where: { isAutoRejected: false }
    });
    
    const foundInQueue = queueItems.some(t => t.id === transcript.id);
    expect(foundInQueue).toBe(false);
    
    // Cleanup
    await prisma.transcript.delete({ where: { id: transcript.id } });
    await prisma.audioFile.delete({ where: { id: audioFile.id } });
  });
  
  test('Audio >= 15s is NOT auto-rejected', async () => {
    const audioFile = await prisma.audioFile.create({
      data: {
        filename: 'long_20sec.wav',
        path: '/test/long_20sec.wav',
        duration: 20.0, // >= 15 seconds
        sampleRate: 16000,
        channels: 1,
        bitDepth: 16,
        fileSize: 640000
      }
    });
    
    const transcript = await prisma.transcript.create({
      data: {
        audioFileId: audioFile.id,
        originalText: 'This is a longer clip',
        correctedText: 'This is a longer clip',
        isAutoRejected: false, // Should NOT be rejected
        status: 'pending'
      }
    });
    
    expect(transcript.isAutoRejected).toBe(false);
    expect(transcript.status).toBe('pending');
    
    // Should be in queue
    const queueItems = await prisma.transcript.findMany({
      where: { isAutoRejected: false }
    });
    
    const foundInQueue = queueItems.some(t => t.id === transcript.id);
    expect(foundInQueue).toBe(true);
    
    // Cleanup
    await prisma.transcript.delete({ where: { id: transcript.id } });
    await prisma.audioFile.delete({ where: { id: audioFile.id } });
  });
  
  test('Exactly 15 seconds is NOT auto-rejected', async () => {
    const audioFile = await prisma.audioFile.create({
      data: {
        filename: 'exactly_15sec.wav',
        path: '/test/exactly_15sec.wav',
        duration: 15.0, // Exactly 15 seconds
        sampleRate: 16000,
        channels: 1,
        bitDepth: 16,
        fileSize: 480000
      }
    });
    
    const transcript = await prisma.transcript.create({
      data: {
        audioFileId: audioFile.id,
        originalText: 'Exactly fifteen seconds',
        correctedText: 'Exactly fifteen seconds',
        isAutoRejected: false,
        status: 'pending'
      }
    });
    
    expect(transcript.isAutoRejected).toBe(false);
    
    // Cleanup
    await prisma.transcript.delete({ where: { id: transcript.id } });
    await prisma.audioFile.delete({ where: { id: audioFile.id } });
  });
});
```

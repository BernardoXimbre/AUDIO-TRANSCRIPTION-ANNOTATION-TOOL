## ⚡ Exactly

Create `tests/unit/unitNormalization.test.ts`:

```typescript
import { describe, test, expect } from '@jest/globals';
import { normalizeUnit } from '../../src/services/annotationService';

describe('Unit Normalization', () => {
  
  test('mg to g conversion', () => {
    const result = normalizeUnit(1500, 'mg');
    expect(result).toEqual({ value: 1.5, unit: 'g' });
  });
  
  test('ug to g conversion', () => {
    const result = normalizeUnit(1000, 'ug');
    expect(result).toEqual({ value: 0.001, unit: 'g' });
  });
  
  test('kg to g conversion', () => {
    const result = normalizeUnit(5, 'kg');
    expect(result).toEqual({ value: 5000, unit: 'g' });
  });
  
  test('ml to l conversion', () => {
    const result = normalizeUnit(2000, 'ml');
    expect(result).toEqual({ value: 2, unit: 'l' });
  });
  
  test('mm to cm conversion', () => {
    const result = normalizeUnit(10, 'mm');
    expect(result).toEqual({ value: 1, unit: 'cm' });
  });
  
  test('Already normalized units return unchanged', () => {
    const result1 = normalizeUnit(100, 'g');
    expect(result1).toEqual({ value: 100, unit: 'g' });
    
    const result2 = normalizeUnit(50, 'l');
    expect(result2).toEqual({ value: 50, unit: 'l' });
    
    const result3 = normalizeUnit(30, 'cm');
    expect(result3).toEqual({ value: 30, unit: 'cm' });
  });
  
  test('mmHg and IE remain unchanged', () => {
    const result1 = normalizeUnit(120, 'mmHg');
    expect(result1).toEqual({ value: 120, unit: 'mmHg' });
    
    const result2 = normalizeUnit(5000, 'IE');
    expect(result2).toEqual({ value: 5000, unit: 'IE' });
  });
  
  test('Decimal values handled correctly', () => {
    const result = normalizeUnit(0.5, 'kg');
    expect(result.value).toBeCloseTo(500, 1);
    expect(result.unit).toBe('g');
  });
});
```

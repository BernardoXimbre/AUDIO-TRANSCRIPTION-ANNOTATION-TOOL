```typescript
import { describe, test, expect } from '@jest/globals';
import { autoPair, applyManualOverrides } from '../../src/services/pairingService';

describe('Pairing Logic', () => {
  
  test('Auto-pairing matches files by base name', () => {
    const audios = ['sample.wav', 'test.mp3', 'other.m4a'];
    const transcripts = ['sample.json', 'test.json'];
    
    const result = autoPair(audios, transcripts);
    
    expect(result.matched).toHaveLength(2);
    expect(result.matched).toContainEqual({ audio: 'sample.wav', transcript: 'sample.json' });
    expect(result.matched).toContainEqual({ audio: 'test.mp3', transcript: 'test.json' });
    expect(result.unmatchedAudio).toContain('other.m4a');
    expect(result.unmatchedTranscript).toHaveLength(0);
  });
  
  test('Unmatched items detected on both sides', () => {
    const audios = ['a.wav', 'b.mp3', 'c.wav'];
    const transcripts = ['a.json', 'd.json'];
    
    const result = autoPair(audios, transcripts);
    
    expect(result.matched).toHaveLength(1);
    expect(result.unmatchedAudio).toContain('b.mp3');
    expect(result.unmatchedAudio).toContain('c.wav');
    expect(result.unmatchedTranscript).toContain('d.json');
  });
  
  test('Case-insensitive matching', () => {
    const audios = ['Sample.WAV'];
    const transcripts = ['sample.json'];
    
    const result = autoPair(audios, transcripts);
    
    expect(result.matched).toHaveLength(1);
    expect(result.matched[0]).toEqual({ audio: 'Sample.WAV', transcript: 'sample.json' });
  });
  
  test('Manual override pairs unmatched items', () => {
    const initial = {
      matched: [{ audio: 'a.wav', transcript: 'a.json' }],
      unmatchedAudio: ['b.wav', 'c.wav'],
      unmatchedTranscript: ['d.json']
    };
    
    const override = [{ audio: 'c.wav', transcript: 'd.json' }];
    const result = applyManualOverrides(initial, override);
    
    expect(result.matched).toHaveLength(2);
    expect(result.matched).toContainEqual({ audio: 'c.wav', transcript: 'd.json' });
    expect(result.unmatchedAudio).toEqual(['b.wav']);
    expect(result.unmatchedTranscript).toHaveLength(0);
  });
  
  test('No silent drops - all unmatched reported', () => {
    const audios = ['file1.wav', 'file2.mp3', 'file3.m4a'];
    const transcripts = ['file1.json', 'file4.json'];
    
    const result = autoPair(audios, transcripts);
    
    const totalUnmatched = result.unmatchedAudio.length + result.unmatchedTranscript.length;
    expect(totalUnmatched).toBe(3); // file2, file3, file4
  });
});
```

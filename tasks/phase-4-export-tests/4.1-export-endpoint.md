## ⚡ Exactly

Create `apps/backend/src/controllers/exportController.ts`:

```typescript
import { Request, Response } from 'express';
import { prisma } from '../db';

export async function exportJSONL(req: Request, res: Response) {
  try {
    // Query completed transcripts with all related data
    const transcripts = await prisma.transcript.findMany({
      where: { status: 'completed' },
      include: {
        audioFile: {
          select: {
            filename: true,
            duration: true,
            sampleRate: true,
            channels: true,
            bitDepth: true,
            bextMetadata: true
          }
        },
        annotations: true,
        recording: true
      }
    });
    
    // Build JSONL output
    const jsonlLines = transcripts.map(transcript => {
      const record = {
        audioPath: transcript.audioFile.filename,
        duration: transcript.audioFile.duration,
        recordingConditions: {
          sampleRate: transcript.audioFile.sampleRate,
          channels: transcript.audioFile.channels,
          bitDepth: transcript.audioFile.bitDepth,
          speechRate: transcript.recording?.speechRate || null,
          distanceEstimate: transcript.recording?.distanceEstimate || null
        },
        transcript: {
          original: transcript.originalText,
          corrected: transcript.correctedText
        },
        annotations: transcript.annotations.map(ann => ({
          type: ann.type,
          startOffset: ann.startOffset,
          endOffset: ann.endOffset,
          attributes: ann.attributes
        }))
      };
      
      return JSON.stringify(record);
    });
    
    const jsonlContent = jsonlLines.join('\n');
    
    // Return as downloadable file
    res.setHeader('Content-Type', 'application/jsonl');
    res.setHeader('Content-Disposition', 'attachment; filename="export.jsonl"');
    res.send(jsonlContent);
  } catch (error) {
    const message = error instanceof Error ? error.message : String(error);
    res.status(500).json({ error: message });
  }
}
```

Add to Express app:

```typescript
import { exportJSONL } from './controllers/exportController';

app.post('/api/export', exportJSONL);
```

```typescript
import { prisma } from '../db';

export interface QueueFilterOptions {
  status?: 'pending' | 'in_progress' | 'completed';
  minDuration?: number;
  maxDuration?: number;
  sortBy?: 'duration' | 'status' | 'filename' | 'createdAt';
  order?: 'asc' | 'desc';
  offset?: number;
  limit?: number;
}

/**
 * Query queue with filters and sorting
 */
export async function queryQueue(filters: QueueFilterOptions) {
  const {
    status,
    minDuration,
    maxDuration,
    sortBy = 'duration',
    order = 'desc',
    offset = 0,
    limit = 100
  } = filters;
  
  // Build where clause
  const where: any = {
    isAutoRejected: false
  };
  
  if (status) {
    where.status = status;
  }
  
  if (minDuration !== undefined || maxDuration !== undefined) {
    where.audioFile = {
      duration: {}
    };
    if (minDuration !== undefined) {
      (where.audioFile.duration as any).gte = minDuration;
    }
    if (maxDuration !== undefined) {
      (where.audioFile.duration as any).lte = maxDuration;
    }
  }
  
  // Build orderBy
  const orderBy: any = {};
  const sortOrder = order === 'asc' ? 'asc' : 'desc';
  
  switch (sortBy) {
    case 'duration':
      orderBy.audioFile = { duration: sortOrder };
      break;
    case 'status':
      orderBy.status = sortOrder;
      break;
    case 'filename':
      orderBy.audioFile = { filename: sortOrder };
      break;
    case 'createdAt':
    default:
      orderBy.createdAt = sortOrder;
  }
  
  // Query
  const items = await prisma.transcript.findMany({
    where,
    select: {
      id: true,
      status: true,
      audioFile: {
        select: {
          id: true,
          filename: true,
          duration: true
        }
      }
    },
    orderBy,
    skip: offset,
    take: limit
  });
  
  const total = await prisma.transcript.count({ where });
  
  return {
    items,
    total,
    offset,
    limit,
    hasMore: offset + limit < total
  };
}

/**
 * Validate filter options
 */
export function validateFilters(filters: QueueFilterOptions): string[] {
  const errors: string[] = [];
  
  if (filters.status && !['pending', 'in_progress', 'completed'].includes(filters.status)) {
    errors.push('Invalid status. Must be: pending, in_progress, or completed');
  }
  
  if (filters.minDuration !== undefined && filters.minDuration < 0) {
    errors.push('minDuration must be >= 0');
  }
  
  if (filters.maxDuration !== undefined && filters.maxDuration < 0) {
    errors.push('maxDuration must be >= 0');
  }
  
  if (filters.minDuration !== undefined && filters.maxDuration !== undefined && 
      filters.minDuration > filters.maxDuration) {
    errors.push('minDuration must be <= maxDuration');
  }
  
  if (filters.sortBy && !['duration', 'status', 'filename', 'createdAt'].includes(filters.sortBy)) {
    errors.push('Invalid sortBy. Must be: duration, status, filename, or createdAt');
  }
  
  if (filters.order && !['asc', 'desc'].includes(filters.order)) {
    errors.push('Invalid order. Must be: asc or desc');
  }
  
  if (filters.offset !== undefined && filters.offset < 0) {
    errors.push('offset must be >= 0');
  }
  
  if (filters.limit !== undefined && (filters.limit < 1 || filters.limit > 1000)) {
    errors.push('limit must be between 1 and 1000');
  }
  
  return errors;
}
```

```typescript
import { queryQueue, validateFilters } from '../queueService';

describe('queueService', () => {
  test('validateFilters: rejects invalid status', () => {
    const errors = validateFilters({ status: 'invalid' as any });
    expect(errors).toContain(expect.stringContaining('Invalid status'));
  });
  
  test('validateFilters: rejects minDuration > maxDuration', () => {
    const errors = validateFilters({ minDuration: 100, maxDuration: 50 });
    expect(errors).toContain(expect.stringContaining('minDuration'));
  });
  
  test('validateFilters: accepts valid filters', () => {
    const errors = validateFilters({ 
      status: 'pending',
      minDuration: 30,
      maxDuration: 120,
      sortBy: 'duration',
      order: 'asc'
    });
    expect(errors).toHaveLength(0);
  });
});
```

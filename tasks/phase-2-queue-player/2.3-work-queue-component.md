```vue
<template>
  <div class="work-queue max-w-6xl mx-auto p-6">
    <h1 class="text-3xl font-bold mb-2">Work Queue</h1>
    <p class="text-gray-600 mb-6">Click any item to start annotating</p>
    
    <!-- Filters -->
    <div class="flex gap-4 mb-6">
      <div class="flex gap-2">
        <label class="font-semibold">Status:</label>
        <select v-model="filters.status" @change="loadQueue" class="border rounded px-2 py-1">
          <option value="">All</option>
          <option value="pending">Pending</option>
          <option value="in_progress">In Progress</option>
          <option value="completed">Completed</option>
        </select>
      </div>
      
      <div class="flex gap-2">
        <label class="font-semibold">Duration:</label>
        <input 
          v-model.number="filters.minDuration" 
          @change="loadQueue"
          type="number" 
          placeholder="Min (s)" 
          class="border rounded px-2 py-1 w-20"
        />
        <span>-</span>
        <input 
          v-model.number="filters.maxDuration" 
          @change="loadQueue"
          type="number" 
          placeholder="Max (s)" 
          class="border rounded px-2 py-1 w-20"
        />
      </div>
    </div>
    
    <!-- Loading State -->
    <div v-if="loading" class="p-4 bg-blue-50 border border-blue-300 rounded text-blue-700">
      ⏳ Loading queue...
    </div>
    
    <!-- Error State -->
    <div v-if="error" class="p-4 bg-red-50 border border-red-300 rounded text-red-700 mb-4">
      ❌ Error: {{ error }}
    </div>
    
    <!-- Empty State -->
    <div v-if="!loading && items.length === 0" class="p-8 bg-gray-50 border border-gray-300 rounded text-center text-gray-500">
      No items in queue. Upload audio files to get started.
    </div>
    
    <!-- Queue Table -->
    <div v-if="!loading && items.length > 0" class="overflow-x-auto mb-6">
      <table class="w-full border-collapse text-sm">
        <thead>
          <tr class="bg-gray-200 border-b-2 border-gray-400">
            <th class="px-4 py-2 text-left">Filename</th>
            <th class="px-4 py-2 text-left">Duration (s)</th>
            <th class="px-4 py-2 text-left">Status</th>
            <th class="px-4 py-2 text-left">
              <button 
                @click="cycleSortBy('createdAt')"
                class="hover:text-blue-600"
              >
                Created {{ sortBy === 'createdAt' ? (order === 'asc' ? '↑' : '↓') : '' }}
              </button>
            </th>
            <th class="px-4 py-2 text-left">Action</th>
          </tr>
        </thead>
        <tbody>
          <tr 
            v-for="item in items"
            :key="item.id"
            class="border-b hover:bg-gray-50 cursor-pointer"
            @click="selectItem(item)"
          >
            <td class="px-4 py-2 font-mono text-blue-600">{{ item.audioFile.filename }}</td>
            <td class="px-4 py-2">{{ item.audioFile.duration.toFixed(1) }}</td>
            <td class="px-4 py-2">
              <span :class="statusBadgeClass(item.status)">{{ item.status }}</span>
            </td>
            <td class="px-4 py-2 text-sm text-gray-500">{{ formatDate(item.createdAt) }}</td>
            <td class="px-4 py-2">
              <button 
                @click.stop="openAnnotator(item)"
                class="px-3 py-1 bg-blue-500 text-white text-xs rounded hover:bg-blue-600"
              >
                Annotate →
              </button>
            </td>
          </tr>
        </tbody>
      </table>
    </div>
    
    <!-- Pagination -->
    <div v-if="!loading && items.length > 0" class="flex justify-between items-center">
      <div class="text-sm text-gray-600">
        Showing {{ offset + 1 }}-{{ offset + items.length }} of {{ total }}
      </div>
      <div class="flex gap-2">
        <button 
          @click="previousPage"
          :disabled="offset === 0"
          class="px-3 py-1 border rounded hover:bg-gray-100 disabled:opacity-50"
        >
          ← Previous
        </button>
        <button 
          @click="nextPage"
          :disabled="!hasMore"
          class="px-3 py-1 border rounded hover:bg-gray-100 disabled:opacity-50"
        >
          Next →
        </button>
      </div>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue';
import { useRouter } from 'vue-router';

interface QueueItem {
  id: string;
  status: string;
  audioFile: {
    id: string;
    filename: string;
    duration: number;
  };
  createdAt: string;
}

const router = useRouter();

const items = ref<QueueItem[]>([]);
const loading = ref(false);
const error = ref('');
const total = ref(0);
const offset = ref(0);
const limit = 50;
const hasMore = ref(false);

const filters = ref({
  status: '',
  minDuration: undefined as number | undefined,
  maxDuration: undefined as number | undefined
});

const sortBy = ref('duration');
const order = ref('desc');

async function loadQueue() {
  loading.value = true;
  error.value = '';
  
  try {
    const params = new URLSearchParams();
    if (filters.value.status) params.append('status', filters.value.status);
    if (filters.value.minDuration !== undefined) params.append('minDuration', String(filters.value.minDuration));
    if (filters.value.maxDuration !== undefined) params.append('maxDuration', String(filters.value.maxDuration));
    params.append('sortBy', sortBy.value);
    params.append('order', order.value);
    params.append('offset', String(offset.value));
    params.append('limit', String(limit));
    
    const response = await fetch(\`/api/queue?\${params}\`);
    
    if (!response.ok) {
      throw new Error(\`Failed to load queue: \${response.statusText}\`);
    }
    
    const data = await response.json();
    items.value = data.items;
    total.value = data.total;
    hasMore.value = data.hasMore;
  } catch (err) {
    error.value = err instanceof Error ? err.message : 'Unknown error';
  } finally {
    loading.value = false;
  }
}

function statusBadgeClass(status: string): string {
  const baseClass = 'px-2 py-1 rounded text-xs font-semibold';
  switch (status) {
    case 'pending':
      return \`\${baseClass} bg-yellow-100 text-yellow-700\`;
    case 'in_progress':
      return \`\${baseClass} bg-blue-100 text-blue-700\`;
    case 'completed':
      return \`\${baseClass} bg-green-100 text-green-700\`;
    default:
      return \`\${baseClass} bg-gray-100 text-gray-700\`;
  }
}

function formatDate(dateStr: string): string {
  const date = new Date(dateStr);
  return date.toLocaleDateString() + ' ' + date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
}

function cycleSortBy(field: string) {
  if (sortBy.value === field) {
    order.value = order.value === 'asc' ? 'desc' : 'asc';
  } else {
    sortBy.value = field;
    order.value = 'desc';
  }
  offset.value = 0;
  loadQueue();
}

function selectItem(item: QueueItem) {
  // Store selected item for annotator view
  sessionStorage.setItem('selectedTranscriptId', item.id);
}

function openAnnotator(item: QueueItem) {
  selectItem(item);
  router.push({ name: 'Annotator', params: { transcriptId: item.id } });
}

function previousPage() {
  offset.value = Math.max(0, offset.value - limit);
  loadQueue();
}

function nextPage() {
  if (hasMore.value) {
    offset.value += limit;
    loadQueue();
  }
}

onMounted(() => {
  loadQueue();
});
</script>

<style scoped>
.work-queue {
  background: white;
  min-height: 100vh;
}
</style>
```

Register in router:

```typescript
import WorkQueue from '../components/WorkQueue.vue';

const routes = [
  { path: '/queue', component: WorkQueue, name: 'Queue' },
  // ...
];
```

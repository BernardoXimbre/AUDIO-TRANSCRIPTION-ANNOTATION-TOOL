## ⚡ Exactly

Create `apps/backend/src/controllers/queueController.ts`:

```typescript
import { Request, Response } from 'express';
import { prisma } from '../db';

export async function getQueue(req: Request, res: Response) {
  try {
    const { 
      status, 
      sortBy = 'duration', 
      order = 'desc',
      minDuration,
      maxDuration,
      offset = 0,
      limit = 100
    } = req.query;
    
    // Build where clause (only non-rejected items)
    const where: any = {
      isAutoRejected: false
    };
    
    if (status) {
      where.status = String(status);
    }
    
    // Duration range filter
    if (minDuration || maxDuration) {
      where.audioFile = {
        duration: {}
      };
      if (minDuration) {
        (where.audioFile.duration as any).gte = parseFloat(String(minDuration));
      }
      if (maxDuration) {
        (where.audioFile.duration as any).lte = parseFloat(String(maxDuration));
      }
    }
    
    // Determine sort order
    const orderBy: any = {};
    const sortField = String(sortBy);
    const sortOrder = String(order).toLowerCase() === 'asc' ? 'asc' : 'desc';
    
    switch (sortField) {
      case 'duration':
        orderBy.audioFile = { duration: sortOrder };
        break;
      case 'status':
        orderBy.status = sortOrder;
        break;
      case 'filename':
        orderBy.audioFile = { filename: sortOrder };
        break;
      case 'createdAt':
      default:
        orderBy.createdAt = sortOrder;
    }
    
    // Query transcripts
    const items = await prisma.transcript.findMany({
      where,
      select: {
        id: true,
        status: true,
        isAutoRejected: false,
        audioFile: {
          select: {
            id: true,
            filename: true,
            duration: true,
            path: true,
            sampleRate: true,
            channels: true
          }
        }
      },
      orderBy,
      skip: parseInt(String(offset), 10),
      take: parseInt(String(limit), 10)
    });
    
    // Get total count
    const total = await prisma.transcript.count({ where });
    
    res.status(200).json({
      items,
      total,
      offset: parseInt(String(offset), 10),
      limit: parseInt(String(limit), 10)
    });
  } catch (error) {
    const message = error instanceof Error ? error.message : String(error);
    res.status(500).json({ error: message });
  }
}
```

Add to Express app (`apps/backend/src/server.ts`):

```typescript
import { getQueue } from './controllers/queueController';

app.get('/api/queue', getQueue);
```
